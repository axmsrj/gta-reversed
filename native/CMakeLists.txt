cmake_minimum_required(VERSION 4.2)

# This target intentionally has no dependency on the injected Windows target.
if(NOT CMAKE_HOST_SYSTEM_NAME STREQUAL "Darwin")
    message(FATAL_ERROR "GTA San Andreas Native requires macOS on Apple Silicon")
endif()
set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Supported architecture" FORCE)
set(CMAKE_OSX_DEPLOYMENT_TARGET 26.0 CACHE STRING "Minimum macOS version" FORCE)
project(GTASanAndreasNative VERSION 0.1.0 LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)

option(GTASA_NATIVE_SANITIZERS "Enable address and undefined behavior checks" OFF)
if(GTASA_NATIVE_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()

add_executable(gta_sa_native MACOSX_BUNDLE Main.mm PlayerController.cpp)
target_compile_options(gta_sa_native PRIVATE -fobjc-arc -Wall -Wextra -Wpedantic)
target_link_libraries(gta_sa_native PRIVATE "-framework Cocoa" "-framework Metal" "-framework QuartzCore")
set_target_properties(gta_sa_native PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER "org.gtasanandreas.native"
    MACOSX_BUNDLE_BUNDLE_NAME "GTA San Andreas Native"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in")

include(CTest)
if(BUILD_TESTING)
    add_executable(player_controller_tests tests/PlayerControllerTests.cpp PlayerController.cpp)
    target_include_directories(player_controller_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME player_controller_behavior COMMAND player_controller_tests)
    add_test(NAME native_metal_smoke COMMAND gta_sa_native --smoke-frames 3)
    set_tests_properties(native_metal_smoke PROPERTIES TIMEOUT 30 LABELS "gpu;smoke")
    add_test(NAME native_reject_bad_frame_count COMMAND gta_sa_native --smoke-frames invalid)
    set_tests_properties(native_reject_bad_frame_count PROPERTIES WILL_FAIL TRUE TIMEOUT 5)
endif()

option(GTASA_WITH_LIBRW_PROBE "Build the real DFF/TXD parsing probe" ON)
if(GTASA_WITH_LIBRW_PROBE)
    set(LIBRW_PLATFORM NULL CACHE STRING "Headless RenderWare parsing backend" FORCE)
    set(LIBRW_TOOLS OFF CACHE BOOL "No upstream viewer tools" FORCE)
    set(LIBRW_INSTALL OFF CACHE BOOL "No install from private port build" FORCE)
    set(GTASA_LIBRW_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../local/librw"
        CACHE PATH "Pinned librw source checkout")
    if(EXISTS "${GTASA_LIBRW_SOURCE_DIR}/CMakeLists.txt")
        add_subdirectory("${GTASA_LIBRW_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/librw")
    else()
        include(FetchContent)
        FetchContent_Declare(librw
            GIT_REPOSITORY https://github.com/aap/librw.git
            GIT_TAG 18532c2e13efbc1aa43f0b00b4459831f9414de0)
        FetchContent_MakeAvailable(librw)
    endif()
    add_executable(gta_sa_rw_probe assets/ProbeRw.cpp)
    target_link_libraries(gta_sa_rw_probe PRIVATE gta_sa_asset_core librw)
    target_link_libraries(gta_sa_native PRIVATE gta_sa_asset_core gta_sa_collision_core gta_sa_animation_core librw)
    target_link_libraries(gta_sa_native PRIVATE gta_sa_world_core)
    target_compile_definitions(gta_sa_native PRIVATE GTASA_HAS_LIBRW=1)
endif()
